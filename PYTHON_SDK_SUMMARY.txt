from pap import (
    PrincipalKeypair,
    Scope,
    ScopeAction,
    DisclosureSet,
    Mandate,
)
from datetime import datetime, timezone, timedelta

# Generate keypair
principal = PrincipalKeypair.generate()

# Issue mandate
scope = Scope([ScopeAction("schema:SearchAction")])
ttl = (datetime.now(timezone.utc) + timedelta(hours=24)).isoformat()

mandate = Mandate.issue_root(
    principal.did(),
    "did:key:zagent",
    scope,
    DisclosureSet.empty(),
    ttl,
)

mandate.sign(principal)
mandate.verify_with_keypair(principal)  # ✓
