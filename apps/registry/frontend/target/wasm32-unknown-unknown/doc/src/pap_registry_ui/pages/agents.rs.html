9#[component]
10pub fn AgentsPage() -> impl IntoView {
11    let query = RwSignal::new(String::new());
12    let page = RwSignal::new(1u32);
13    let show_register = RwSignal::new(false);
14    // Incrementing this forces a re-fetch (used by Refresh button and on_remove).
15    let refresh_key = RwSignal::new(0u32);
16
17    // create_resource tracks (query, page, refresh_key) as its source and re-fetches
18    // automatically when any of them changes — one fetch per source change, no double-fetches.
19    let agents = create_resource(
20        move || (query.get(), page.get(), refresh_key.get()),
21        |(q, pg, _)| async move {
22            let q_ref = if q.is_empty() { None } else { Some(q.as_str()) };
23            api::fetch_agents(q_ref, pg, PER_PAGE).await
24        },
25    );
26
27    let reload = move || refresh_key.update(|n| *n += 1);
28
29    view! {
30        <div class="page">
31            <div class="page-header">
32                <div style="display:flex; align-items:center; justify-content:space-between">
33                    <div>
34                        <h1 class="page-title">"Agents"</h1>
35                        <p class="page-subtitle">"All registered agent advertisements in this node's registry."</p>
36                    </div>
37                    <button class="btn btn-primary" on:click=move |_| show_register.set(true)>
38                        "+ Register Agent"
39                    </button>
40                </div>
41            </div>
42
43            <div class="filter-bar">
44                <input
45                    class="filter-input"
46                    placeholder="Search by name, provider, or capability…"
47                    prop:value=move || query.get()
48                    on:input=move |e| {
49                        let val = event_target_value(&e);
50                        // batch ensures query + page reset fire as one reactive update
51                        batch(move || {
52                            query.set(val);
53                            page.set(1);
54                        });
55                    }
56                />
57                <button class="btn btn-secondary" on:click=move |_| reload()>
58                    "↺ Refresh"
59                </button>
60            </div>
61
62            {move || match agents.get() {
63                None => view! { <div class="loading">"Loading agents…"</div> }.into_any(),
64                Some(Err(e)) => view! {
65                    <div class="error-banner">"Error: " {e}</div>
66                }.into_any(),
67                Some(Ok(resp)) => {
68                    if resp.items.is_empty() {
69                        view! {
70                            <div class="empty-state">
71                                <div class="empty-state-icon">"⬡"</div>
72                                <div class="empty-state-title">"No agents found"</div>
73                                <p style="font-size:13px; color: var(--text-3)">
74                                    {if query.get().is_empty() {
75                                        "No agents are registered yet. Click \"Register Agent\" to add one."
76                                    } else {
77                                        "No agents match your search."
78                                    }}
79                                </p>
80                            </div>
81                        }.into_any()
82                    } else {
83                        let total = resp.total;
84                        let total_pages = resp.total_pages;
85                        view! {
86                            <div>
87                                <div class="agent-list">
88                                    {resp.items.into_iter().map(|entry| view! {
89                                        <AgentCard entry=entry.clone() on_remove=move || reload() />
90                                    }).collect::<Vec<_>>()}
91                                </div>
92                                <div class="pagination">
93                                    <button
94                                        class="btn btn-secondary btn-sm"
95                                        disabled=move || page.get() <= 1
96                                        on:click=move |_| page.update(|p| *p -= 1)
97                                    >
98                                        "← Prev"
99                                    </button>
100                                    <span class="pagination-info">
101                                        "Page " {move || page.get()} " of " {total_pages}
102                                        " (" {total} " total)"
103                                    </span>
104                                    <button
105                                        class="btn btn-secondary btn-sm"
106                                        disabled=move || page.get() >= total_pages
107                                        on:click=move |_| page.update(|p| *p += 1)
108                                    >
109                                        "Next →"
110                                    </button>
111                                </div>
112                            </div>
113                        }.into_any()
114                    }
115                }
116            }}
117
118            {move || if show_register.get() {
119                view! {
120                    <RegisterModal
121                        on_close=move || show_register.set(false)
122                        on_success=move || { show_register.set(false); reload(); }
123                    />
124                }.into_any()
125            } else {
126                view! { <span /> }.into_any()
127            }}
128        </div>
129    }
130}

132#[component]
133fn AgentCard(
134    entry: AgentEntry,
135    #[prop(into)] on_remove: Callback<()>,
136) -> impl IntoView {
137    let hash = entry.hash.clone();
138    let hash_display = {
139        let h = &hash;
140        if h.len() > 16 { format!("{}…", &h[..16]) } else { h.clone() }
141    };
142    let ad = entry.ad;
143    let name = ad.name.clone();
144    let provider_name = ad.provider.name.clone();
145    let provider_did = ad.provider.did.clone();
146    let capabilities = ad.capability.clone();
147    let returns = ad.returns.clone();
148    let disclosure = ad.requires_disclosure.clone();
149    let signed_by = ad.signed_by.clone();
150
151    let removing = RwSignal::new(false);
152    let error = RwSignal::new(None::<String>);
153
154    let do_remove = {
155        let hash = hash.clone();
156        move |_| {
157            let hash = hash.clone();
158            removing.set(true);
159            spawn_local(async move {
160                match api::remove_agent(&hash).await {
161                    Ok(()) => on_remove.run(()),
162                    Err(e) => {
163                        error.set(Some(e));
164                        removing.set(false);
165                    }
166                }
167            });
168        }
169    };
170
171    view! {
172        <div class="agent-card">
173            <div style="display:flex; justify-content:space-between; align-items:flex-start">
174                <div style="flex:1; min-width:0">
175                    <div class="agent-name">{name}</div>
176                    <div class="agent-provider">
177                        {provider_name} " · " {provider_did}
178                    </div>
179                </div>
180                <button
181                    class="btn btn-danger btn-sm"
182                    disabled=move || removing.get()
183                    on:click=do_remove
184                >
185                    {move || if removing.get() { "…" } else { "Remove" }}
186                </button>
187            </div>
188
189            {move || error.get().map(|e| view! {
190                <div class="error-banner" style="margin-top: var(--sp-xs)">{e}</div>
191            })}
192
193            <div class="agent-meta">
194                {capabilities.iter().map(|c| view! {
195                    <span class="tag tag-action">{c.clone()}</span>
196                }).collect::<Vec<_>>()}
197                {returns.iter().map(|r| view! {
198                    <span class="tag tag-returns">{r.clone()}</span>
199                }).collect::<Vec<_>>()}
200                {disclosure.iter().map(|d| view! {
201                    <span class="tag tag-disclosure">{d.clone()}</span>
202                }).collect::<Vec<_>>()}
203            </div>
204
205            <div class="agent-did">"Signed by: " {signed_by}</div>
206            <div class="agent-hash">"Hash: " {hash_display}</div>
207        </div>
208    }
209}

211#[component]
212fn RegisterModal(
213    #[prop(into)] on_close: Callback<()>,
214    #[prop(into)] on_success: Callback<()>,
215) -> impl IntoView {
216    let json_input = RwSignal::new(String::new());
217    let error = RwSignal::new(None::<String>);
218    let submitting = RwSignal::new(false);
219
220    let do_submit = move |_| {
221        let json = json_input.get();
222        if json.trim().is_empty() {
223            error.set(Some("Advertisement JSON is required.".into()));
224            return;
225        }
226        submitting.set(true);
227        error.set(None);
228        spawn_local(async move {
229            match serde_json::from_str::<crate::api::AgentAdvertisement>(&json) {
230                Err(e) => {
231                    error.set(Some(format!("Invalid JSON: {}", e)));
232                    submitting.set(false);
233                }
234                Ok(_) => {
235                    let window = web_sys::window().unwrap();
236                    let mut opts = web_sys::RequestInit::new();
237                    opts.set_method("POST");
238                    opts.set_mode(web_sys::RequestMode::SameOrigin);
239                    opts.set_body(&wasm_bindgen::JsValue::from_str(&json));
240                    match web_sys::Request::new_with_str_and_init("/api/agents", &opts) {
241                        Err(e) => {
242                            error.set(Some(format!("Request error: {:?}", e)));
243                            submitting.set(false);
244                        }
245                        Ok(req) => {
246                            let _ = req.headers().set("Content-Type", "application/json");
247                            match wasm_bindgen_futures::JsFuture::from(
248                                window.fetch_with_request(&req),
249                            )
250                            .await
251                            {
252                                Err(e) => {
253                                    error.set(Some(format!("Fetch error: {:?}", e)));
254                                    submitting.set(false);
255                                }
256                                Ok(resp_val) => {
257                                    let resp: web_sys::Response =
258                                        resp_val.dyn_into().unwrap();
259                                    if resp.ok() {
260                                        on_success.run(());
261                                    } else {
262                                        error.set(Some(format!(
263                                            "Server error: HTTP {}",
264                                            resp.status()
265                                        )));
266                                        submitting.set(false);
267                                    }
268                                }
269                            }
270                        }
271                    }
272                }
273            }
274        });
275    };
276
277    view! {
278        <div class="modal-backdrop">
279            <div class="modal">
280                <div class="modal-title">"Register Agent Advertisement"</div>
281                <div class="form-group">
282                    <label class="form-label">"Advertisement JSON"</label>
283                    <textarea
284                        class="form-textarea"
285                        placeholder=PLACEHOLDER_AD
286                        prop:value=move || json_input.get()
287                        on:input=move |e| json_input.set(event_target_value(&e))
288                    />
289                    <div class="form-hint">
290                        "Paste a signed AgentAdvertisement JSON-LD object. Must include a valid Ed25519 signature."
291                    </div>
292                </div>
293
294                {move || error.get().map(|e| view! {
295                    <div class="error-banner">{e}</div>
296                })}
297
298                <div class="modal-actions">
299                    <button class="btn btn-secondary" on:click=move |_| on_close.run(())>
300                        "Cancel"
301                    </button>
302                    <button
303                        class="btn btn-primary"
304                        disabled=move || submitting.get()
305                        on:click=do_submit
306                    >
307                        {move || if submitting.get() { "Registering…" } else { "Register" }}
308                    </button>
309                </div>
310            </div>
311        </div>
312    }
313}
