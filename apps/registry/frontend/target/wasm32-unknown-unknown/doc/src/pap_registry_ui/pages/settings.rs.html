6#[component]
7pub fn SettingsPage() -> impl IntoView {
8    let status = RwSignal::new(None::<crate::api::RegistryStatus>);
9    let loading = RwSignal::new(true);
10
11    Effect::new(move || {
12        spawn_local(async move {
13            if let Ok(s) = api::fetch_status().await {
14                status.set(Some(s));
15            }
16            loading.set(false);
17        });
18    });
19
20    view! {
21        <div class="page">
22            <div class="page-header">
23                <h1 class="page-title">"Settings"</h1>
24                <p class="page-subtitle">"Registry configuration and node identity."</p>

26
27            <div class="card" style="margin-bottom: var(--sp-xl)">
28                <div class="card-header">
29                    <span class="card-title">"Node Identity"</span>
30                </div>
31                <div class="card-body">
32                    {move || if loading.get() {
33                        view! { <div class="loading">"Loading…"</div> }.into_any()
34                    } else if let Some(s) = status.get() {
35                        view! {
36                            <div>
37                                <div class="form-group">
38                                    <label class="form-label">"DID"</label>
39                                    <div class="identity-did">{s.did.clone()}</div>
40                                </div>
41                                <div class="form-group">
42                                    <label class="form-label">"TLS Certificate Fingerprint"</label>

44                                        {s.cert_fingerprint.clone()}

47                                        "Share this fingerprint with peers so they can pin this node's TLS certificate."
48                                    </div>
49                                </div>
50                                <div class="form-group">
51                                    <label class="form-label">"Public Endpoint"</label>
52                                    <div style="font-family: var(--font-mono); font-size: 13px; color: var(--blue); background: var(--bg-2); padding: var(--sp-sm) var(--sp-md); border-radius: var(--r-md)">
53                                        {s.endpoint.clone()}
54                                    </div>
55                                    <div class="form-hint">
56                                        "Set via PAP_REGISTRY_ENDPOINT environment variable."
57                                    </div>
58                                </div>
59                            </div>
