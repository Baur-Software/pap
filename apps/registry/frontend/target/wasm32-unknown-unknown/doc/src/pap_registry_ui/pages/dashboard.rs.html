1use leptos::prelude::*;
2use wasm_bindgen_futures::spawn_local;
3
4use crate::api::{self, RegistryStatus};

6#[component]
7pub fn DashboardPage() -> impl IntoView {
8    let status = RwSignal::new(None::<RegistryStatus>);
9    let error = RwSignal::new(None::<String>);
10    let loading = RwSignal::new(true);
11
12    Effect::new(move || {
13        spawn_local(async move {
14            match api::fetch_status().await {
15                Ok(s) => {
16                    status.set(Some(s));
17                    error.set(None);
18                }
19                Err(e) => error.set(Some(e)),
20            }
21            loading.set(false);
22        });
23    });
24
25    view! {
26        <div class="page">
27            <div class="page-header">
28                <h1 class="page-title">"Dashboard"</h1>
29                <p class="page-subtitle">"Federation node overview and registry health."</p>
30            </div>
31
32            {move || error.get().map(|e| view! {
33                <div class="error-banner">"Failed to load status: " {e}</div>
34            })}
35
36            {move || if loading.get() {
37                view! { <div class="loading"><span class="loading-dot">"Loading..."</span></div> }.into_any()
38            } else if let Some(s) = status.get() {
39                view! {
40                    <div>
41                        <div class="stat-grid">
42                            <div class="stat-card">
43                                <div class="stat-label">"Registered Agents"</div>
44                                <div class="stat-value teal">{s.agent_count}</div>
45                            </div>
46                            <div class="stat-card">
47                                <div class="stat-label">"Federation Peers"</div>
48                                <div class="stat-value accent">{s.peer_count}</div>
49                            </div>
50                            <div class="stat-card">
51                                <div class="stat-label">"Version"</div>
52                                <div class="stat-value" style="font-size: 20px; font-family: var(--font-mono)">{s.version.clone()}</div>
53                            </div>
54                        </div>
55
56                        <div class="identity-panel">
57                            <div class="stat-label">"Node Identity"</div>
58                            <div class="identity-did">{s.did.clone()}</div>
59                            <div class="identity-fp">
60                                <span style="color: var(--text-3)">"TLS fingerprint: "</span>
61                                {s.cert_fingerprint.clone()}
62                            </div>
63                            <div style="margin-top: var(--sp-sm); font-size: 13px; color: var(--text-2)">
64                                "Public endpoint: "
65                                <span style="font-family: var(--font-mono); color: var(--blue)">{s.endpoint.clone()}</span>
66                            </div>
67                        </div>
68
69                        <div class="card">
70                            <div class="card-header">
71                                <span class="card-title">"Federation Endpoints"</span>
72                            </div>
73                            <div class="card-body">
74                                <p style="font-size: 13px; color: var(--text-2); margin-bottom: var(--sp-md)">
75                                    "This node exposes the PAP federation protocol. Other registries can connect at:"
76                                </p>
77                                <EndpointRow method="GET"  path="/federation/identity" desc="Node identity and cert fingerprint" />
78                                <EndpointRow method="GET"  path="/federation/query?action=..." desc="Query agents by Schema.org action" />
79                                <EndpointRow method="POST" path="/federation/announce" desc="Receive agent advertisement" />
80                                <EndpointRow method="GET"  path="/federation/peers" desc="Known peer list" />
81                            </div>
82                        </div>
83                    </div>
84                }.into_any()
85            } else {
86                view! { <div class="empty-state"><div class="empty-state-title">"No data"</div></div> }.into_any()
87            }}
88        </div>
89    }
90}

92#[component]
93fn EndpointRow(
94    method: &'static str,
95    path: &'static str,
96    desc: &'static str,
97) -> impl IntoView {
98    let method_color = if method == "GET" { "var(--teal)" } else { "var(--gold)" };
99    view! {
100        <div style="display:flex; gap: var(--sp-md); align-items: baseline; padding: var(--sp-xs) 0; border-bottom: 1px solid var(--border-subtle)">
101            <span style=format!("font-family: var(--font-mono); font-size: 11px; font-weight: 600; color: {}; width: 36px; flex-shrink: 0", method_color)>
102                {method}
103            </span>
104            <span style="font-family: var(--font-mono); font-size: 12px; color: var(--purple); flex: 1">
105                {path}
106            </span>
107            <span style="font-size: 12px; color: var(--text-3)">{desc}</span>
108        </div>
109    }
